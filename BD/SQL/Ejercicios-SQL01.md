### JOIN

```SQL
SELECT P.pname , D.dname FROM PRO P JOIN DEPT D ON P.deptno = D.deptno
```

```SQL
SELECT E.ename , EP.prono FROM EMP E JOIN EMPPRO EP ON E.empno  = EP.empno
```

```SQL
SELECT E.ename , EP.prono FROM EMP E FULL JOIN EMPPRO EP ON E.empno  = EP.empno
```

```SQL
SELECT E.ename , J.empno FROM EMP E LEFT JOIN EMP J ON E.mgr = J.empno
```

```SQL
SELECT E1.ename , E2.ename, D.dname FROM EMP E1 LEFT JOIN EMP E2 ON E1.mgr = E2.empno JOIN DEPT D ON E2.deptno = D.deptno
```

```SQL
SELECT SUB.ename FROM EMP SUB JOIN EMP JEF ON SUB.mgr = JEF.empno AND SUB.SAL > JEF.SAL
```

```SQL
SELECT E.ENAME, COALESCE(SUM(EP.HOURS), 0) AS HORAS FROM EMP E LEFT JOIN EMPPRO EP ON EP.EMPNO = E.EMPNO GROUP BY E.EMPNO, E.ENAME;
```

```SQL
SELECT J.EMPNO, J.ENAME, COUNT(S.EMPNO) AS SUBORD_GANAN_MAS FROM EMP J LEFT JOIN EMP S
ON S.MGR = J.EMPNO AND S.SAL > J.SAL
WHERE J.EMPNO IN (SELECT MGR FROM EMP WHERE MGR IS NOT NULL)
GROUP BY J.EMPNO, J.ENAME;
```
